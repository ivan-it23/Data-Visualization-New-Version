CAN_setBitTiming(CANA_BASE, 11, 0, 7, 6, 3);//5 11
//

        Interrupt_initVectorTable();

        // Interrupts that are used in this example are re-mapped to ISR functionsd

        //
        Interrupt_register(INT_XINT1, &xint1ISR);
        //
        // Assign the interrupt service routines to ePWM interrupts

        Interrupt_register(INT_EPWM6, &epwm6ISR);

        // Disable sync(Freeze clock to PWM as well)

        initEPWM6_2000_kHz();

        initEPWM8_533_kHz();

        //
        Interrupt_enable(INT_CANA0);
        CAN_enableGlobalInterrupt(CANA_BASE, CAN_GLOBAL_INT_CANINT0);
        // Enable CAN test mode with external loopback

        CAN_disableTestMode(CANA_BASE);
        //
        // Initialize the transmit message object used for sending CAN messages.
        // Message Object Parameters:
        //      CAN Module: A
        //      Message Object ID Number: 1
        //      Message Identifier: 0x95555555
        //      Message Frame: Extended
        //      Message Type: Transmit
        //      Message ID Mask: 0x0
        //      Message Object Flags: None
        //      Message Data Length: 4 Bytes

        CAN_setupMessageObject(CANA_BASE, TX_MSG_OBJ_ID, ID_transmitter, CAN_MSG_FRAME_STD, CAN_MSG_OBJ_TYPE_TX, 0, CAN_MSG_OBJ_TX_INT_ENABLE, MSG_DATA_LENGTH);

        //      CAN Module: B
        //      Message Object ID Number: 1

        //      Message Frame: Extended
        //      Message Type: Receive
        //      Message ID Mask: 0x0
        //      Message Object Flags: Receive Interrupt
        //      Message Data Length: 4 Bytes
        //
        // Инициализируем CAN на прием сообщения от 1-го передатчика:
        CAN_setupMessageObject(CANA_BASE, RX_MSG_OBJ_ID_Tx1, 0x311, CAN_MSG_FRAME_STD, CAN_MSG_OBJ_TYPE_RX, 0, CAN_MSG_OBJ_RX_INT_ENABLE, MSG_DATA_LENGTH);

        // Инициализируем CAN на прием сообщения от 3-го передатчика:
        CAN_setupMessageObject(CANA_BASE, RX_MSG_OBJ_ID_Tx3, 0x313, CAN_MSG_FRAME_STD, CAN_MSG_OBJ_TYPE_RX, 0, CAN_MSG_OBJ_RX_INT_ENABLE, MSG_DATA_LENGTH);

        //
        // Configure XINT1 to be a triggered by a falling edge and XINT2 to be

        //Выход XCLKOUT - клок DDS 48 MHz
        // GPIO18->Reserve (N/A for GPIO)

        //

        GPIO_setDirectionMode(DEVICE_GPIO_PIN_SCITXDA, GPIO_DIR_MODE_OUT);

        //

        //
        DINT;
        //
        // Map the ISR to the wake interrupt.
        //
        //Interrupt_register(INT_SCIA_TX, sciaTxISR);
        Interrupt_register(INT_SCIA_RX, sciaRxISR);
        //
        // Initialize SCIA and its FIFO.
        //
        SCI_performSoftwareReset(SCIA_BASE);
        //
        // Configure SCIA for echoback.
        //

                                                SCI_CONFIG_PAR_NONE));

        SCI_clearInterruptStatus(SCIA_BASE, SCI_INT_RXFF);
        SCI_enableFIFO(SCIA_BASE);
        SCI_enableModule(SCIA_BASE);
        SCI_performSoftwareReset(SCIA_BASE);
        //
        // Set the transmit FIFO level to 0 and the receive FIFO level to 2.
        // Enable the TXFF and RXFF interrupts.
        //
        SCI_setFIFOInterruptLevel(SCIA_BASE, SCI_FIFO_TX0, SCI_FIFO_RX5);
        SCI_enableInterrupt(SCIA_BASE, SCI_INT_RXFF);
        //
        // Clear the SCI interrupts before enabling them.
        //
        SCI_clearInterruptStatus(SCIA_BASE, SCI_INT_RXFF);
        //
        // Enable the interrupts in the PIE: Group 9 interrupts 1 & 2.
        //
        Interrupt_enable(INT_SCIA_RX);
        Interrupt_clearACKGroup(INTERRUPT_ACK_GROUP9);
        //
        // Enable global interrupts.
        //
        EINT;
        //
        DEVICE_DELAY_US(10000); // 100ms (approx.) pause before starting the job
        //
        ad9956_Init();
        ad9956_Read(0x00, from1DataCFR1, 4);
        for(i=0; i<4; i++) {
         if(from1DataCFR1[i]!=to1DataCFR1[i]) {
           readEqualWrite = false;
           break;
         }
        }
        ad9956_Read(0x01, from1DataCFR2, 5);
        for(i=0; i<5; i++) {
         if(from1DataCFR2[i]!=to1DataCFR2[i]) {
           readEqualWrite = false;
           break;
         }
        }
        ad9956_Read(0x06, from1DataPCR0, 8);
        for(i=0; i<8; i++) {
         if(from1DataPCR0[i]!=to1DataPCR0[i]) {
           readEqualWrite = false;
           break;
         }
        }
        //Устанавливаем частоту 2МГц в профиль 4
        //0xAAC 0000 0000 //AAA 4B27 2CEE
        DDSinit_dat[0]=0x04;//0x04; // номер профиля
        DDSinit_dat[1]=0x0A;//0x0A;//0x0A;//0x0A;//0x0A;//0x09;//0x0A;//0x05; // частота ст
        DDSinit_dat[2]=0x74;//0x74;//0x3E;//0x74;//0xAC;//0xD9;//0xAC;//0x1E;
        DDSinit_dat[3]=0x45;//0x45;//0xC5;//0x45;//0x00;//0xF2;//0x00;//0xB8;
        DDSinit_dat[4]=0xA8;//0xA8;//0xF9;//0xA8;//0x00;//0xDF;//0x00;//0xB6; // частота мл

//
// sciaRxISR - Read two characters from the RXBUF and echo them back.`
//
    __interrupt void sciaRxISR(void)
    {
    uint16_t i, j, bytesCounter;
    uint16_t bytesReaded=0;
    uint16_t countDataBytes; // число байтов данных в команде
    uint32_t sciTimePeriod = HWREG(CPUTIMER1_BASE + CPUTIMER_O_PRD)>>1; // переменная для таймаута
    uint16_t bytesToRead = (uint16_t)SCI_getRxFIFOStatus(SCIA_BASE);
    bool errorFlag=false;

    // пакет с командой 5 байт 0х01 - отдать данные последнего измерения.
    // | uint16_t cmd | uint16_t Length l | uint16_t Length H | uint16_t CRC16 L | uint16_t CRC16 H |

    // пакет с командой 5 байт 0х02 - отдать файл метрологии.
    // | uint16_t cmd | uint16_t Length l | uint16_t Length H | uint16_t CRC16 L | uint16_t CRC16 H |

    // пакет с командой 245 байт 0х03 - записать во flash файл метрологии.
    // | uint16_t cmd | uint16_t Length l | uint16_t Length H | 240 байт данных | uint16_t CRC16 L | uint16_t CRC16 H |

    CPUTimer_reloadTimerCounter(CPUTIMER1_BASE);

    for(bytesCounter=0; bytesCounter<bytesToRead; bytesCounter++) {  // заметьте, bytesCounter не уничтожается после цикла, а используется далее в следующем цикле
          rdataB[bytesCounter]=SCI_readCharBlockingFIFO(SCIA_BASE);  // Read data
          bytesReaded++;
        }

    // складываем принятую команду в структуру (без вызова функции parser, так как из обработчика прерывания):
       currCommand.cmdByte = rdataB[0]; bytesReaded--;
       currCommand.Length = (((rdataB[2])<<8) & 0xFF00) | ((rdataB[1]) & 0x00FF); bytesReaded-=2;
       countDataBytes = currCommand.Length - 5;
       /*currCommand.CRC = (((rdataB[244])<<8) & 0xFF00) | ((rdataB[243]) & 0x00FF);*/ bytesReaded-=2;

       // Здесь надо сверить принятую длину команды с тем количеством байтов, которые приняты.
       // 5 байт - это минимальная длина пакета.
       // Если принято меньшее количество байт, то рассовать принятые байты по местам, а затем проверить еще раз на прием байтов пакета.

       // байты данных рассовываем по принципу: пока есть куда класть, пока есть принятые данные, пока не превышен таймаут ожидания данных
           j=0;
           while(j<countDataBytes) {
             if(bytesReaded>0) {
               currCommand.DataBytes[j] = rdataB[5+j];
               bytesReaded--;
               j++;
             }else {
               bytesToRead=(uint16_t)SCI_getRxFIFOStatus(SCIA_BASE);
               for(i=0; i<bytesToRead; i++) {
                 rdataB[bytesCounter++]=SCI_readCharBlockingFIFO(SCIA_BASE);
                 bytesReaded++;
               }
             }
             if( CPUTimer_getTimerCount(CPUTIMER1_BASE) < sciTimePeriod ) {
               errorFlag = true;
               // выход по таймауту
               break;
             }
           }



   if(currCommand.Length>5){
   //currCommand.CRC = (((rdataB[244])<<8) & 0xFF00) | ((rdataB[243]) & 0x00FF);
   currCommand.CRC = (((rdataB[currCommand.Length-1])<<8) & 0xFF00) | ((rdataB[currCommand.Length-2]) & 0x00FF);
   }
   else{
   currCommand.CRC = (((rdataB[4])<<8) & 0xFF00) | ((rdataB[3]) & 0x00FF);
   }
   currCommand.CalculatedCRC = inlineGetCRC16(rdataB,currCommand.Length-2);
   //currCommand.CRC = (((rdataB[244])<<8) & 0xFF00) | ((rdataB[243]) & 0x00FF);
   //
   // Clear the SCI RXFF interrupt and acknowledge the PIE interrupt.
   //


   if(errorFlag == false) {
   //  currCommand.done = false;
     commandReceived = true;
   }else {
   //  currCommand.done = false;
     commandReceived = false;
   }


    SCI_clearInterruptStatus(SCIA_BASE, SCI_INT_RXFF);
    Interrupt_clearACKGroup(INTERRUPT_ACK_GROUP9);

    }



    void initCPUTimers(void)
    {
    //
    // Initialize timer period to maximum
    //
    CPUTimer_setPeriod(CPUTIMER0_BASE, 0xFFFFFFFF);
    //
    // Initialize pre-scale counter to divide by 1 (SYSCLKOUT)
    //
    CPUTimer_setPreScaler(CPUTIMER0_BASE, 0);
    //
    // Make sure timer is stopped
    //
    CPUTimer_stopTimer(CPUTIMER0_BASE);
    //
    // Reload all counter register with period value
    //
    CPUTimer_reloadTimerCounter(CPUTIMER0_BASE);
    //
    // Reset interrupt counter
    //
    cpuTimer0IntCount = 0;

    }
    //
    // configCPUTimer - This function initializes the selected timer to the
    // period specified by the "freq" and "period" parameters. The "freq" is
    // entered as Hz and the period in uSeconds. The timer is held in the stopped
    // state after configuration.
    //
    void configCPUTimer(uint32_t cpuTimer, float freq, float period)
    {
    uint32_t temp;
    //
    // Initialize timer period:
    //
    temp = (uint32_t)(freq / 1000000 * period);
    CPUTimer_setPeriod(cpuTimer, temp);
    //
    // Set pre-scale counter to divide by 1 (SYSCLKOUT):
    //
    CPUTimer_setPreScaler(cpuTimer, 0);
    //
    // Initializes timer control register. The timer is stopped, reloaded,
    // free run disabled, and interrupt enabled.
    // Additionally, the free and soft bits are set
    //
    CPUTimer_stopTimer(cpuTimer);
    CPUTimer_reloadTimerCounter(cpuTimer);
    CPUTimer_setEmulationMode(cpuTimer, CPUTIMER_EMULATIONMODE_STOPAFTERNEXTDECREMENT);
    CPUTimer_enableInterrupt(cpuTimer);
    //
    // Resets interrupt counters for the three cpuTimers
    //
    if (cpuTimer == CPUTIMER0_BASE)
    {
        cpuTimer0IntCount = 0;
    }

    }

    //
    // cpuTimer0ISR - Counter for CpuTimer0
    //
    __interrupt void cpuTimer0ISR(void)
    {
    uint16_t i;


       cpuTimer0IntCount++; // счетчик прерываний самого таймера

       for(i=0; i<NCntrs; i++) {
         if((usCntr[i]>0)&&(usCntrEnable[i]==true)) {
           usCntr[i]-=usPeriod;
         }
       }


    if(cpuTimer0IntCount==500000){//500000
        cpuTimer0IntCount = 0;
    }
    //
    // Acknowledge this interrupt to receive more interrupts from group 1
    //
    Interrupt_clearACKGroup(INTERRUPT_ACK_GROUP1);

    usAutoDelayCntr-=usPeriod;


    if( (usCntr[1]<=0) && (currProgramMode==WorkCyclogramState)) {
        usCntr[1]=2097152;
    }

    }

    void initSPIB()
    {
     //
     // GPIO30 is the SPIB SIMO pin.
     //
     GPIO_setMasterCore(30, GPIO_CORE_CPU1);
     GPIO_setPinConfig(GPIO_30_SPISIMOB);
     GPIO_setDirectionMode(30, GPIO_DIR_MODE_OUT);
     GPIO_setPadConfig(30, GPIO_PIN_TYPE_STD);
     GPIO_setQualificationMode(30, GPIO_QUAL_ASYNC);
     //
     // GPIO31 is the SPIB SOMI pin.
     //
     GPIO_setMasterCore(31, GPIO_CORE_CPU1);
     GPIO_setPinConfig(GPIO_31_SPISOMIB);
     GPIO_setDirectionMode(31, GPIO_DIR_MODE_IN);
     GPIO_setPadConfig(31, GPIO_PIN_TYPE_STD);
     GPIO_setQualificationMode(31, GPIO_QUAL_ASYNC);
     //
     // GPIO14 is the SPIB CLK pin.
     //
     GPIO_setMasterCore(14, GPIO_CORE_CPU1);
     GPIO_setPinConfig(GPIO_14_SPICLKB);
     GPIO_setDirectionMode(14, GPIO_DIR_MODE_OUT);
     GPIO_setPadConfig(14, GPIO_PIN_TYPE_STD);
     GPIO_setQualificationMode(14, GPIO_QUAL_ASYNC);
     //
     // GPIO15 is the SPIB STE (CS для DDS) pin.
     //
     GPIO_setMasterCore(15, GPIO_CORE_CPU1);
     GPIO_setPinConfig(GPIO_15_SPISTEB);
     GPIO_setDirectionMode(15, GPIO_DIR_MODE_OUT);
     GPIO_setPadConfig(15, GPIO_PIN_TYPE_STD);
     GPIO_setQualificationMode(15, GPIO_QUAL_ASYNC);
     //
     // Must put SPI into reset before configuring it.
     //
     SPI_disableModule(SPIB_BASE);
     //
     // SPI configuration. Use a 2MHz SPICLK and 8-bit word size.
     //
     SPI_setConfig(SPIB_BASE, DEVICE_LSPCLK_FREQ, SPI_PROT_POL1PHA0,
                   SPI_MODE_MASTER, 2500000, 8); // 5000000
     //
     // Configuration complete. Enable the module.
     //
     SPI_enableModule(SPIB_BASE);
    }

    //
    // CAN B ISR - The interrupt service routine called when a CAN interrupt is
    //             triggered on CAN module B.
    //
    __interrupt void canaISR(void)
    {
    uint32_t status;
    //
    // Read the CAN-B interrupt status to find the cause of the interrupt
    //
    status = CAN_getInterruptCause(CANA_BASE);
    //
    // If the cause is a controller status interrupt, then get the status
    //
    if(status == CAN_INT_INT0ID_STATUS)
    {
    //
    // Read the controller status.  This will return a field of status
    // error bits that can indicate various errors.  Error processing
    // is not done in this example for simplicity.  Refer to the
    // API documentation for details about the error status bits.
    // The act of reading this status will clear the interrupt.
    //
    status = CAN_getStatus(CANA_BASE);

    if(((status  & ~(CAN_STATUS_TXOK | CAN_STATUS_RXOK)) != 7) && ((status  & ~(CAN_STATUS_TXOK | CAN_STATUS_RXOK)) != 0)){
                    g_bErrFlag++;
                    if(g_bErrFlag<maxTryCount) {
                        txInProgress = true;
                        if(TX_MSG_OBJ_ID==1)
                            //CANMessageSet(CANA_BASE, currTxObj, &sTXCANMessage1, MSG_OBJ_TYPE_TX); сообщение на отправку как у Глеба, у нас ф-ция на отправук другая:
                            CAN_sendMessage(CANA_BASE, TX_MSG_OBJ_ID, MSG_DATA_LENGTH,txMsgData);
                        //
                        // Set a flag to indicate some errors may have occurred.
                        //
                        // errorFlag = 1;
                } else  txInProgress = false;
            }
    }
    //
    // Check if the cause is the CAN-B receive message object 1
    //
    else if(status == RX_MSG_OBJ_ID_Tx1)
    {
        CAN_readMessage(CANA_BASE, RX_MSG_OBJ_ID_Tx1, rxMsgData[RX_MSG_OBJ_ID_Tx1]);
        CAN_clearInterruptStatus(CANA_BASE, RX_MSG_OBJ_ID_Tx1);
        g_ulRxMsgCount[2]++;
        errorFlag = 0;/// ну в будующем будем что-то с эти  флагом делать
    }
    else if(status == RX_MSG_OBJ_ID_Tx2)
    {
        CAN_readMessage(CANA_BASE, RX_MSG_OBJ_ID_Tx2, rxMsgData[RX_MSG_OBJ_ID_Tx2]);
        CAN_clearInterruptStatus(CANA_BASE, RX_MSG_OBJ_ID_Tx2);
        g_ulRxMsgCount[3]++;
        errorFlag = 0;/// ну в будующем будем что-то с эти  флагом делать
    }
    else if(status == RX_MSG_OBJ_ID_Tx3)
    {
        CAN_readMessage(CANA_BASE, RX_MSG_OBJ_ID_Tx3, rxMsgData[RX_MSG_OBJ_ID_Tx3]);
        CAN_clearInterruptStatus(CANA_BASE, RX_MSG_OBJ_ID_Tx3);
        g_ulRxMsgCount[4]++;
        errorFlag = 0;/// ну в будующем будем что-то с эти  флагом делать
    }
    else if(status == RX_MSG_OBJ_ID_Tx4)
    {
        CAN_readMessage(CANA_BASE, RX_MSG_OBJ_ID_Tx4, rxMsgData[RX_MSG_OBJ_ID_Tx4]);
        CAN_clearInterruptStatus(CANA_BASE, RX_MSG_OBJ_ID_Tx4);
        g_ulRxMsgCount[5]++;
        errorFlag = 0;/// ну в будующем будем что-то с эти  флагом делать
    }
    else if(status == RX_MSG_OBJ_ID_Tx5)
    {
        CAN_readMessage(CANA_BASE, RX_MSG_OBJ_ID_Tx5, rxMsgData[RX_MSG_OBJ_ID_Tx5]);
        CAN_clearInterruptStatus(CANA_BASE, RX_MSG_OBJ_ID_Tx5);
        g_ulRxMsgCount[6]++;
        errorFlag = 0;/// ну в будующем будем что-то с эти  флагом делать
    }

    else if(status == TX_MSG_OBJ_ID)
        {

            txInProgress = false;
            //
            // Getting to this point means that the RX interrupt occurred on
            // message object 1, and the message RX is complete.  Clear the
            // message object interrupt.
            //
            CAN_clearInterruptStatus(CANA_BASE, TX_MSG_OBJ_ID);

            //
            // Increment a counter to keep track of how many messages have been
            // received. In a real application this could be used to set flags to
            // indicate when a message is received.
            //
            //rxMsgCount++;
            //
            // Since the message was received, clear any error flags.
            //
            errorFlag = 0;/// ну в будующем будем что-то с эти  флагом делать
        }
        //
        // If something unexpected caused the interrupt, this would handle it.
        //
        else
        {
        //
        // Spurious interrupt handling can go here.
        //
        }
        //
        // Clear the global interrupt flag for the CAN interrupt line
        //
        CAN_clearGlobalInterruptStatus(CANA_BASE, CAN_GLOBAL_INT_CANINT0);
        //
        // Acknowledge this interrupt located in group 9
        //
        Interrupt_clearACKGroup(INTERRUPT_ACK_GROUP9);
    }

    //

    //
    // Configure the SOC to occur on the first up-count event
    //
    EPWM_setADCTriggerSource(EPWM1_BASE, EPWM_SOC_A, EPWM_SOC_TBCTR_U_CMPA);
    EPWM_setADCTriggerEventPrescale(EPWM1_BASE, EPWM_SOC_A, 1);

    EPWM_setTimeBasePeriod(EPWM1_BASE, 0x800);
    //
    // Freeze the counter
    //
    EPWM_setTimeBaseCounterMode(EPWM1_BASE, EPWM_COUNTER_MODE_STOP_FREEZE);
    EPWM_enableADCTrigger(EPWM1_BASE, EPWM_SOC_A);
    EPWM_setTimeBaseCounterMode(EPWM1_BASE, EPWM_COUNTER_MODE_UP);

    }

                // делаем из фаз приемников разность фаз, приведенную к +- PI

            float d_ph(float ph2, float ph1){
                float result;
                result = ph2 - ph1;
                if (fabs(result) < 40) {
                    while (result > M_PI) result -= 2 * M_PI;
                    while (result <= -M_PI) result += 2 * M_PI;
                }
                else result = 0.0f;
                return result;
                }

            //прямая задача
            //n_Tx - номер передатчика
            //вспомогательная функция для расчета УЭС выполняется для одного зонда внутри функции RO_ARG
                float SIGNAL(struct GP_METROLOGY *metro, uint16_t k_Tx, uint16_t freq, float ro) {
                complex float re = 1.0f;//комплексное число с 0-й мнимой частью. 1.0f- воспринемаеться как вещественное число.
                float L1 = metro->L1[k_Tx] / 1000.0;
                float L2 = metro->L2[k_Tx] / 1000.0;
                float omegamu0sigma = (0.0078957 * metro->F[freq]) / ro;
                complex float ii = _Imaginary_I;
                complex float ik = ii*csqrtf(omegamu0sigma*_Imaginary_I);///заменяем I на ii т.к в cjmplex.h нет ф-ции умножения комп.чисел, результат 0.0
                complex float SGN = cexpf(ik*(L2 - L1))*((re - ik * L2)/(re - ik * L1));
                return cargf(SGN);
            }

                void RO_ARG(struct GP_METROLOGY *metro, struct GP_DATA *Data, uint16_t k_Tx, uint16_t freq) {
                float Ro0 = 0.0f;
                float epsilon_ARG = 0.0000005; // точность  фазы
                float epsilon_INF = 0.000000005; // точность  фазы
                float  Ro_0 = 0.01, Ro_max = 7000.0;  //мин. и мах. значение ус для расчета
                bool stop_epsl_ARG = false;
                bool stop_epsl_INF = false;
                float  ro_0 = Ro_0;
                float  ro_max = Ro_max;

                do {
                    float X1 = ro_0 + 0.382*(ro_max - ro_0);
                    float X2 = ro_max - 0.382*(ro_max - ro_0);
                    float A = 0;
                    float B = 0;




                    A = Data->phase_smt[freq][k_Tx] - SIGNAL(metro, k_Tx, freq, X1);//
                    B = Data->phase_smt[freq][k_Tx] - SIGNAL(metro, k_Tx, freq, X2);//

                    if (fabs(A) > fabs(B)) { ro_0 = X1; }
                    else { ro_max = X2; }
                    if (fabs(A) < epsilon_ARG || fabs(B) < epsilon_ARG) { stop_epsl_ARG = true; }
                    else if (fabs(A - B) < epsilon_INF) { stop_epsl_INF = true; }

                } while (stop_epsl_ARG == false && stop_epsl_INF == false);
                if (stop_epsl_ARG == true) Ro0 = (ro_0 + ro_max) / 2;
                if (stop_epsl_INF == true) Ro0 = 7200;//-100
                ro_0 = Ro_0;
                ro_max = Ro_max;
                Data->rho_smt[freq][k_Tx] = Ro0;

                }

            //N_Tx - количество передатчиков
                void simmetry(struct GP_DATA *Data, struct GP_METROLOGY *metro, uint16_t N_Tx) {
                    //выполняем для двух частот
                    enum T_ {
                    T1, T2, T3, T4, T5
                };

                    for (int freq = 0; freq < 2; freq++){
                        uint16_t condition;
                        float d_PH[5] = { 0.0, };
                        float K[5][5];

                        for (int Tx = 0; Tx < N_Tx; Tx++) {
                            d_PH[Tx] = Data->DELTA_PH[freq][Tx];
                        }

                        // для автоматического воздуха
                        for (int Tx = 0; Tx < N_Tx; Tx++){
                            //скользящее среднее  для воздуха
                            air_summ[Tx + 5*freq] -= air_buff[w_ma][Tx + 5*freq];
                            air_buff[w_ma][Tx + 5*freq] = d_PH[Tx];
                            air_summ[Tx + 5*freq] += air_buff[w_ma][Tx + 5*freq];
                        }
                        // нормализуем сигнал к воздуху
                        //Z1 повернут T1 phRx2-phRx1
                        d_PH[T1] = +(d_PH[T1] - ((float)(metro->Air_zz[freq][T1])) / 57297.0);
                        d_PH[T2] = -(d_PH[T2] - ((float)(metro->Air_zz[freq][T2])) / 57297.0);
                        d_PH[T3] = +(d_PH[T3] - ((float)(metro->Air_zz[freq][T3])) / 57297.0);
                        d_PH[T4] = -(d_PH[T4] - ((float)(metro->Air_zz[freq][T4])) / 57297.0);
                        d_PH[T5] = +(d_PH[T5] - ((float)(metro->Air_zz[freq][T5])) / 57297.0);
                        // перекладываем 32 бит Data->condition в 16 бит condition, в зависимости от частоты
                        // GP_DATA.uint32_t condition;
                        //                  400  kGz  2000 kGz
                        //00000000 00000000 00012345 00012345
                        if(freq == 0){
                            condition = (Data->condition >> 8);
                        }
                        if(freq == 1){
                            uint16_t buff = (Data->condition << 8);
                            condition = (buff >> 8);
                        }

                        //рассчитываем коэф. симметризации в зависимости от работоспособности передатчиков
                        formula_simmetry(K, condition, N_Tx);
                        //производим симметризацию и записываем в структуру Data
                        for (int Tx = 0; Tx < N_Tx; Tx++) {
                            Data->phase_smt[freq][Tx] = K[Tx][T1] * d_PH[T1] + K[Tx][T2] * d_PH[T2] + K[Tx][T3] * d_PH[T3] + K[Tx][T4] * d_PH[T4] + K[Tx][T5] * d_PH[T5];
                        }
                    }
                    //приращение счетчика для скользящего среднего один раз за вызов функции
                    w_ma++;
                    if(w_ma == air_aver){
                        w_ma = 0;
                    }

                }

            //далее  freq = 0 для 400_kGz или freq = 1 для 2000_kGz
            //N_Tx - количество передатчиков
            // condition пишется в GP_DATA.uint32_t condition
                void get_condition(struct GP_DATA *Data, struct GP_METROLOGY *metro, uint16_t N_Tx){

                    // GP_DATA.uint32_t condition;
                    //                  400  kGz  2000 kGz
                    //00000000 00000000 00054321 00054321
                        for (int freq = 0; freq < 2; freq++){
                            uint16_t condition = 0;
                            //Tx1:
                            if (Data->AM_RX_2[freq][T1]  > metro->Air_zz_amt[freq][T1])
                                condition |= (1 << T1);
                            //Tx2:
                            if (Data->AM_RX_1[freq][T2]  > metro->Air_zz_amt[freq][T2])
                                condition |= (1 << T2);
                            //Tx3:
                            if (Data->AM_RX_2[freq][T3] > metro->Air_zz_amt[freq][T3])
                                condition |= (1 << T3);
                            //Tx4:
                            if(N_Tx == 4 || N_Tx == 5){//изменено
                                if (Data->AM_RX_1[freq][T4]  > metro->Air_zz_amt[freq][T4])
                                condition |= (1 << T4);
                            }
                            //Tx5:
                            if(N_Tx == 5){
                                if (Data->AM_RX_2[freq][T5]  > metro->Air_zz_amt[freq][T5])
                                    condition |= (1 << T5);
                            }

                            if(freq == 0)
                                Data->condition |= (condition << 8);
                            if(freq == 1)
                                Data->condition |= (condition << 0);
                        }

        struct ID get_sonde_id(uint32_t signature){
               struct ID tool;
               uint32_t buff;
               buff = signature << 12;
               buff = buff >> 12;
               tool.type_ = buff / 100000;
               tool.N_Tx = (buff % 100000) / 10000;
               tool.mod = (buff % 10000) / 1000;
               tool.number = (buff % 1000);
               tool.type = buff / 1000;
               return tool;
           }

    void calc_silent_Tx(int32_t CH1[12][64], int32_t CH2[12][64], struct GP_DATA *result){

    Q_1 = 0.0; I_1 = 0.0; Q_2 = 0.0; I_2 = 0.0; PH_1 = 0.0; PH_2 = 0.0;
    AMP_1 = 0.0; AMP_2 = 0.0; AMP8_1 = 0.0; AMP8_2 = 0.0;

        }

    for (int i = 0; i < 64; i++) {
            Q_2 += CH2[6][i] * sin_f_64[i]; //второй канал ДФТ для маасивов молчащих передатчиков 400 кГц

    AMP_2 = sqrt(I_2 * I_2 + Q_2 * Q_2);

    PH_2 = atan2f(Q_2 , I_2);

    }

    // Работа по циклограмме:
    void WorkCyclogram()
    {

    static uint16_t TXNumber;
    static uint16_t FreqProfile;
    uint16_t dataIndex;
//    uint16_t j = 0;


    switch (workCyclogramSubState) {
      case WCSSBegin: {
          // обнуляем ошибки передач данных:
          g_ulRxErrCount[0]=0;
          g_ulRxErrCount[1]=0;
          g_ulRxErrCount[2]=0;
          g_ulRxErrCount[3]=0;
          g_ulRxErrCount[4]=0;
          g_ulRxErrCount[5]=0;
          g_ulRxErrCount[6]=0;
          g_ulRxErrCount[7]=0;
          g_ulRxErrCount[8]=0;
          g_ulRxErrCount[9]=0;
          g_ulRxErrCount[10]=0;
          g_ulRxErrCount[11]=0;
          g_ulRxErrCount[12]=0;
          g_ulRxErrCount[13]=0;
          g_ulRxErrCount[14]=0;

         usCntr[0]=1000;
         workCyclogramSubState = WCSSWaitInclinoResponse;
         break;
      }
      // Выполнение накопления данных на частоте 400кГц:
      case WCSSWaitInclinoResponse: {
          if( usCntr[0]<=0 ) {
            //молчащие передатчики
            // Устанавливаем частоту 400кГц на DDS,
            DDSinit_dat[0]=0x00;
            ad9956_SetProfile(DDSinit_dat);
            ad9956_Update();
            // на PWM, 400 кгц
            initEPWM6_400_kHz();
            TXNumber = 0; // это значит - слушаем молчащие передатчики
            FreqProfile = 0; // это значит - передаем частоту 400кГц
            GPIO_writePin(testPIN, 1);
            GPIO_enableInterrupt(GPIO_INT_XINT1);// включаем приемник на прием, разрешаем прерывание DRY.
            usCntr[0]=StMDelay;
            workCyclogramSubState = WCSSWaitTXCycle;
          }
          break;
      }

      case WCSSWaitTXCycle: {
          if( usCntr[0]<=0 ) {      // По окончании накопления данных молчащих передатчиков:
              TXNumber++;
              // Посылаем команду на включение передатчика:
              g_ulRxMsgCount[9+TXNumber]=0;
              txMsgData[0] = FreqProfile;
              txMsgData[1] = 0x0000;
              txMsgData[2] = 0x0000;
              txMsgData[3] = 0x0000;
              txMsgData[4] = 0x0000;
              txMsgData[5] = 0x0000;
              txMsgData[6] = 0x0000;
              txMsgData[7] = 0x0000;

              CAN_setupMessageObject(CANA_BASE, TX_MSG_OBJ_ID, 0x210+TXNumber, CAN_MSG_FRAME_STD, CAN_MSG_OBJ_TYPE_TX, 0, CAN_MSG_OBJ_TX_INT_ENABLE, MSG_DATA_LENGTH);

              txInProgress = true;
              g_bErrFlag = 0;
              currTxObj = 1;
              CAN_sendMessage(CANA_BASE, TX_MSG_OBJ_ID, MSG_DATA_LENGTH,txMsgData);
              while(txInProgress) ;
              //Если передали, запускаем ожидание ответа от передатчика
              usCntr[0]=1000;
              workCyclogramSubState = WCSSWaitTXResponse;
          }
          break;
      }

      case WCSSWaitTXResponse: {
          if( usCntr[0]<=0 ) {
            dataIndex = 1+TXNumber;
            if(g_ulRxMsgCount[dataIndex]==1 ) { // передатчик ответил температурой:
               g_ulRxMsgCount[dataIndex]=0;
               g_ulRxErrCount[dataIndex]=0;
               // Сохраняем данные от передатчика
               tmpDataI = (((int32_t)(rxMsgData[dataIndex][0]))<<24)|(((int32_t)(rxMsgData[dataIndex][2]))<<16)|(((int32_t)(rxMsgData[dataIndex][2]))<<8)|((int32_t)(rxMsgData[dataIndex][3]));
               expTxData[TXNumber-1].mcu_temp = (uint16_t)tmpDataI;
            }else { // Неудача! Передатчик промолчал до таймаута!
               g_ulRxMsgCount[dataIndex]=0;
               g_ulRxErrCount[dataIndex]++;
               expTxData[TXNumber-1].mcu_temp = 0;
            }
            usCntr[0]=StMDelay;
            if(TXNumber<=5) {
              // включаем приемник на прием.
              GPIO_writePin(testPIN, 1);
             // GPIO_writePin(testPIN_1, 1);
              GPIO_enableInterrupt(GPIO_INT_XINT1);
              workCyclogramSubState = WCSSWaitTXCycle;
            }else {
              TXNumber = 0;
              if(FreqProfile==0) {
                // Устанавливаем частоту 2000кГц на DDS,
                  DDSinit_dat[0]=0x04;
                  ad9956_SetProfile(DDSinit_dat);
                  ad9956_Update();
                  // на PWM, 2000 кГц
                  initEPWM6_2000_kHz();
                 // включаем приемник на прием.
                  // DEVICE_DELAY_US(10000);
                  GPIO_writePin(testPIN, 1);
